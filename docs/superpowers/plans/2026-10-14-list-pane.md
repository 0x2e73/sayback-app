# List Pane Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the meeting list the left pane and the navigation: search in the drag band, folder chips, meetings grouped by date, a footer with settings and « Démarrer l’enregistrement »; the right card shows the selected meeting, the settings, the recording screen or an empty state. The `Sidebar` and the Library page disappear.

**Architecture:** Pure, Node-tested functions carry the logic (`groupByDate`, `formatRowDate`, `normalizeSearch`, `filterMeetings`, `resolveScope`, `emptyList`, `resolveSelection`, `adjacentId`, `isFindShortcut`). Presentational components in a new `features/list/` folder render them (`MeetingRow`, `MeetingList`, `FolderFilter`, `ListPane`), reusing the existing folder hooks, which move there. `App` keeps one `useWorkspace` hook that owns the selection and the card view; the selection is derived (`resolveSelection`), so the first load and a deletion fall back to the most recent meeting without an effect. The main process relays native full screen over a validated IPC channel; the renderer marks the root with `data-fullscreen`, and the inset tokens drop to 0.

**Tech Stack:** Electron 44, React 19, plain CSS, Vitest 4 (node and jsdom, Testing Library), TypeScript strict with `noUncheckedIndexedAccess`.

**Spec:** `docs/superpowers/specs/2026-10-13-main-window-redesign-design.md`, delivery step 4; sections "Layout", "List pane", "Motion", "Tokens", "Components".

```ts
import type { Meeting } from '../../../shared/contracts';

export interface DateGroup<T> {
  key: string;
  label: string;
  items: T[];
}

type Dated = Pick<Meeting, 'createdAt'>;
type Heading = Omit<DateGroup<never>, 'items'>;

const timeFormat = new Intl.DateTimeFormat('fr-CH', { hour: '2-digit', minute: '2-digit' });
const dayFormat = new Intl.DateTimeFormat('fr-CH', { day: 'numeric', month: 'short' });
const fullDayFormat = new Intl.DateTimeFormat('fr-CH', {
  day: 'numeric',
  month: 'short',
  year: 'numeric',
});
const monthFormat = new Intl.DateTimeFormat('fr-CH', { month: 'long', year: 'numeric' });

function startOfDay(date: Date): Date {
  return new Date(date.getFullYear(), date.getMonth(), date.getDate());
}

// Calendar arithmetic, not milliseconds: the day the clocks change lasts 23 or 25 hours.
function daysBefore(date: Date, days: number): Date {
  return new Date(date.getFullYear(), date.getMonth(), date.getDate() - days);
}

// fr-CH weeks start on Monday; getDay() counts from Sunday.
function startOfWeek(day: Date): Date {
  return daysBefore(day, (day.getDay() + 6) % 7);
}

function monthKey(date: Date): string {
  return `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}`;
}

function capitalise(text: string): string {
  return text.charAt(0).toLocaleUpperCase('fr-CH') + text.slice(1);
}

function heading(date: Date, now: Date): Heading {
  const today = startOfDay(now);
  if (date >= today) return { key: 'today', label: 'Aujourd’hui' };
  if (date >= daysBefore(today, 1)) return { key: 'yesterday', label: 'Hier' };
  if (date >= startOfWeek(today)) return { key: 'week', label: 'Cette semaine' };
  if (date >= new Date(now.getFullYear(), now.getMonth(), 1)) {
    return { key: 'month', label: 'Ce mois-ci' };
  }
  return { key: monthKey(date), label: capitalise(monthFormat.format(date)) };
}

/** Newest first. A date ahead of the clock counts as today. */
export function groupByDate<T extends Dated>(items: readonly T[], now: Date): DateGroup<T>[] {
  const sorted = [...items].sort((a, b) => Date.parse(b.createdAt) - Date.parse(a.createdAt));
  const groups: DateGroup<T>[] = [];
  for (const item of sorted) {
    const { key, label } = heading(new Date(item.createdAt), now);
    const last = groups.at(-1);
    if (last?.key === key) last.items.push(item);
    else groups.push({ key, label, items: [item] });
  }
  return groups;
}

export function formatRowDate(createdAt: string, now: Date): string {
  const date = new Date(createdAt);
  const today = startOfDay(now);
  if (date >= today) return timeFormat.format(date);
  if (date >= daysBefore(today, 1)) return 'Hier';
  if (date.getFullYear() === now.getFullYear()) return dayFormat.format(date);
  return fullDayFormat.format(date);
}
```

The headings are monotonic in time (today, yesterday, this week, this month, then months descending), so after the sort each group is contiguous.

`src/renderer/features/list/use-current-date.ts`:

```ts
import { useEffect, useState } from 'react';

/** The current date, refreshed every minute so « Aujourd’hui » rolls over at midnight. */
export function useCurrentDate(intervalMs = 60_000): Date {
  const [now, setNow] = useState(() => new Date());
  useEffect(() => {
    const timer = setInterval(() => setNow(new Date()), intervalMs);
    return () => clearInterval(timer);
  }, [intervalMs]);
  return now;
}
```

import type { Folder, Meeting } from '../../../shared/contracts';

export type FolderScope = { kind: 'all' } | { kind: 'unfiled' } | { kind: 'folder'; id: string };

export const ALL_MEETINGS: FolderScope = { kind: 'all' };
export const UNFILED: FolderScope = { kind: 'unfiled' };

type Filterable = Pick<Meeting, 'title' | 'notes' | 'folderId'>;

export interface EmptyList {
  message: string;
  searchEverywhere: boolean;
}

interface EmptyListInput {
  total: number;
  scope: FolderScope;
  query: string;
  folders: readonly Folder[];
}

/** Case, accent and ligature insensitive, with runs of whitespace collapsed. */
export function normalizeSearch(text: string): string {
  return text
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLocaleLowerCase('fr-CH')
    .replace(/œ/g, 'oe')
    .replace(/æ/g, 'ae')
    .replace(/\s+/g, ' ')
    .trim();
}

function inScope(meeting: Filterable, scope: FolderScope): boolean {
  if (scope.kind === 'all') return true;
  if (scope.kind === 'unfiled') return !meeting.folderId;
  return meeting.folderId === scope.id;
}

function matches(meeting: Filterable, needle: string): boolean {
  if (!needle) return true;
  const summary = meeting.notes?.summary ?? '';
  return normalizeSearch(`${meeting.title} ${summary}`).includes(needle);
}

export function filterMeetings<T extends Filterable>(
  meetings: readonly T[],
  scope: FolderScope,
  query: string,
): T[] {
  const needle = normalizeSearch(query);
  return meetings.filter((meeting) => inScope(meeting, scope) && matches(meeting, needle));
}

/** A folder deleted or renamed away falls back to every meeting. */
export function resolveScope(scope: FolderScope, folders: readonly Folder[]): FolderScope {
  if (scope.kind !== 'folder') return scope;
  return folders.some((folder) => folder.id === scope.id) ? scope : ALL_MEETINGS;
}

export function emptyList({ total, scope, query, folders }: EmptyListInput): EmptyList {
  const typed = query.replace(/\s+/g, ' ').trim();
  if (total === 0) return { message: 'Aucune réunion', searchEverywhere: false };
  if (typed) {
    return { message: `Aucune réunion pour « ${typed} »`, searchEverywhere: scope.kind !== 'all' };
  }
  if (scope.kind === 'unfiled')
    return { message: 'Aucune réunion sans dossier', searchEverywhere: false };
  const folder = scope.kind === 'folder' ? folders.find((item) => item.id === scope.id) : undefined;
  if (!folder) return { message: 'Aucune réunion', searchEverywhere: false };
  return { message: `Aucune réunion dans « ${folder.name} »`, searchEverywhere: false };
}

import type { Meeting } from '../../../shared/contracts';

type Selectable = Pick<Meeting, 'id' | 'createdAt'>;

/** The chosen meeting while it exists, otherwise the most recent one. */
export function resolveSelection<T extends Selectable>(
  meetings: readonly T[],
  chosenId: string | null,
): T | undefined {
  const chosen = meetings.find((meeting) => meeting.id === chosenId);
  if (chosen) return chosen;
  return meetings.reduce<T | undefined>((latest, meeting) => {
    if (!latest || Date.parse(meeting.createdAt) > Date.parse(latest.createdAt)) return meeting;
    return latest;
  }, undefined);
}

/** Up and down in the visible order, stopping at both ends. */
export function adjacentId(
  ids: readonly string[],
  currentId: string | null,
  step: 1 | -1,
): string | null {
  const index = currentId === null ? -1 : ids.indexOf(currentId);
  if (index === -1) return ids[0] ?? null;
  const next = Math.min(Math.max(index + step, 0), ids.length - 1);
  return ids[next] ?? null;
}

import { useEffect, type RefObject } from 'react';
import { platformOf } from '../../platform';

export interface ShortcutKey {
  key: string;
  metaKey: boolean;
  ctrlKey: boolean;
  altKey: boolean;
  shiftKey: boolean;
}

// On macOS, Ctrl+F moves the caret forward in every text field: only Cmd+F is "find".
export function isFindShortcut(event: ShortcutKey, platform: 'mac' | 'other'): boolean {
  if (event.key.toLowerCase() !== 'f' || event.altKey || event.shiftKey) return false;
  if (platform === 'mac') return event.metaKey && !event.ctrlKey;
  return event.ctrlKey && !event.metaKey;
}

export function useFindShortcut(
  field: RefObject<HTMLInputElement | null>,
  platform = platformOf(navigator.platform),
): void {
  useEffect(() => {
    function onKeyDown(event: KeyboardEvent): void {
      const input = field.current;
      if (!input || !isFindShortcut(event, platform)) return;
      event.preventDefault();
      input.focus();
      input.select();
    }
    window.addEventListener('keydown', onKeyDown);
    return () => window.removeEventListener('keydown', onKeyDown);
  }, [field, platform]);
}

import type { Meeting } from '../../../shared/contracts';

// Chunks report their transcription; the notes phase shows a full line.
function analysisRatio(meeting: Meeting): number {
  const total = meeting.audioChunks.length;
  if (total === 0) return 0;
  return meeting.audioChunks.filter((chunk) => chunk.transcript !== null).length / total;
}

export function RowStatus({ meeting }: { meeting: Meeting }) {
  if (meeting.status === 'ready') {
    const summary = meeting.notes?.summary.trim();
    if (!summary) return null;
    return <span className="meeting-row-summary">{summary}</span>;
  }
  return (
    <span className={`row-status row-status-${meeting.status}`}>
      <span className="row-status-label">
        <span className="row-status-dot" aria-hidden="true" />
        {statusLabels[meeting.status]}
      </span>
      {meeting.status === 'processing' && (
        <span className="row-progress" aria-hidden="true">
          <span
            className="row-progress-bar"
            style={{ transform: `scaleX(${analysisRatio(meeting)})` }}
          />
        </span>
      )}
    </span>
  );

import { useState, type DragEvent } from 'react';
import type { Folder, Meeting } from '../../../shared/contracts';
import { formatRowDate } from './date-groups';
import { RowStatus } from './RowStatus';
import { MEETING_DRAG_TYPE } from './useFolderDrop';

interface MeetingRowProps {
  meeting: Meeting;
  folder?: Folder;
  selected: boolean;
  focusable: boolean;
  locked: boolean;
  now: Date;
  onSelect(id: string): void;
}

export function MeetingRow(props: MeetingRowProps) {
  const { meeting, folder, selected, focusable, locked, now, onSelect } = props;
  const [dragging, setDragging] = useState(false);
  // A move while recording or analysing would be overwritten by the final save.
  const settled = meeting.status !== 'recording' && meeting.status !== 'processing';

  function select(): void {
    if (!locked) onSelect(meeting.id);
  }

  function startDrag(event: DragEvent<HTMLDivElement>): void {
    event.dataTransfer.setData(MEETING_DRAG_TYPE, meeting.id);
    event.dataTransfer.effectAllowed = 'move';
    setDragging(true);
  }

  return (
    <div
      role="option"
      aria-selected={selected}
      aria-disabled={locked || undefined}
      tabIndex={focusable ? 0 : -1}
      data-meeting-id={meeting.id}
      className={`meeting-row${dragging ? ' is-dragging' : ''}`}
      draggable={!locked && settled}
      onClick={select}
      onDragStart={startDrag}
      onDragEnd={() => setDragging(false)}
    >
      <span className="meeting-row-heading">
        <span className="meeting-row-title">{meeting.title}</span>
        <time className="meeting-row-date" dateTime={meeting.createdAt}>
          {formatRowDate(meeting.createdAt, now)}
        </time>
      </span>
      <RowStatus meeting={meeting} />
      {folder && <span className="chip chip-static">{folder.name}</span>}
    </div>
  );
}

const steps: Partial<Record<string, 1 | -1>> = { ArrowDown: 1, ArrowUp: -1 };

function focusOption(list: HTMLElement, id: string): void {
  for (const option of list.querySelectorAll<HTMLElement>('[role="option"]')) {
    if (option.dataset.meetingId === id) option.focus();
  }
}

export function MeetingList(props: MeetingListProps) {
  const { groups, folders, selectedId, now, locked, onSelect } = props;
  const baseId = useId();
  const ids = groups.flatMap((group) => group.items.map((meeting) => meeting.id));
  const tabStop = selectedId !== null && ids.includes(selectedId) ? selectedId : ids[0];

  function handleKeyDown(event: KeyboardEvent<HTMLDivElement>): void {
    const step = steps[event.key];
    if (!step || locked) return;
    event.preventDefault();
    const next = adjacentId(ids, selectedId, step);
    if (!next || next === selectedId) return;
    onSelect(next);
    focusOption(event.currentTarget, next);
  }

  return (
    <div role="listbox" aria-label="Réunions" className="meeting-list" onKeyDown={handleKeyDown}>
      {groups.map((group) => (
        <div
          key={group.key}
          role="group"
          aria-labelledby={`${baseId}-${group.key}`}
          className="date-group"
        >
          <div id={`${baseId}-${group.key}`} className="date-group-heading" aria-hidden="true">
            {group.label}
          </div>
          {group.items.map((meeting) => (
            <MeetingRow
              key={meeting.id}
              meeting={meeting}
              folder={folders.find((folder) => folder.id === meeting.folderId)}
              selected={meeting.id === selectedId}
              focusable={meeting.id === tabStop}
              locked={locked}
              now={now}
              onSelect={onSelect}
            />
          ))}
        </div>
      ))}
    </div>
  );
}

import { useEffect, useRef, type FocusEvent, type KeyboardEvent, type RefObject } from 'react';
import type { Folder } from '../../../shared/contracts';

interface FolderMenuProps {
  folder: Folder;
  anchor: RefObject<HTMLButtonElement | null>;
  onRename(folder: Folder): void;
  onDelete(folder: Folder): void;
  onClose(restoreFocus: boolean): void;
}

function moveFocus(menu: HTMLElement, step: 1 | -1): void {
  const items = [...menu.querySelectorAll<HTMLElement>('[role="menuitem"]')];
  const index = items.findIndex((item) => item === document.activeElement);
  items[(index + step + items.length) % items.length]?.focus();
}

export function FolderMenu({ folder, anchor, onRename, onDelete, onClose }: FolderMenuProps) {
  const first = useRef<HTMLButtonElement>(null);
  useEffect(() => {
    first.current?.focus();
  }, []);

  function handleKeyDown(event: KeyboardEvent<HTMLDivElement>): void {
    if (event.key === 'Escape') {
      event.preventDefault();
      onClose(true);
      return;
    }
    if (event.key !== 'ArrowDown' && event.key !== 'ArrowUp') return;
    event.preventDefault();
    moveFocus(event.currentTarget, event.key === 'ArrowDown' ? 1 : -1);
  }

  // Focus leaving for the chip is left to the chip's click, which toggles the menu shut.
  function handleBlur(event: FocusEvent<HTMLDivElement>): void {
    const next = event.relatedTarget;
    if (next === anchor.current) return;
    if (next instanceof Node && event.currentTarget.contains(next)) return;
    onClose(false);
  }

  return (
    <div
      className="folder-menu"
      role="menu"
      aria-label={folder.name}
      onKeyDown={handleKeyDown}
      onBlur={handleBlur}
    >
      <button ref={first} type="button" role="menuitem" onClick={() => onRename(folder)}>
        Renommer
      </button>
      <button type="button" role="menuitem" onClick={() => onDelete(folder)}>
        Supprimer…
      </button>
    </div>
  );
}

    event.preventDefault();
    onSave();

    event.preventDefault();
    onCancel();

        aria-label="Nom du dossier"
        placeholder="Nom du dossier"
        maxLength={80}
        value={name}
        readOnly={busy}
        onChange={(event) => onName(event.target.value)}
        onKeyDown={handleKeyDown}
        onBlur={() => {
          if (!name.trim()) onCancel();

    setMenuOpen(false);
    onScope(next);
  }

    setMenuOpen(false);
    if (restoreFocus) menuAnchor.current?.focus();

  function rename(folder: Folder): void {
    setMenuOpen(false);
    editor.startRename(folder);
  }

  function requestDeletion(folder: Folder): void {
    setMenuOpen(false);
    deletion.request(folder);
  }

  return (
    <div className="folder-filter">
      <FolderChips
        folders={folders}
        scope={scope}
        locked={locked}
        menuOpen={menuOpen}
        menuAnchor={menuAnchor}
        dropProps={dropProps}
        onChoose={choose}
        onToggleMenu={() => setMenuOpen((open) => !open)}
        onCreate={editor.startCreate}
      />
      {menuOpen && selected && (
        <FolderMenu
          folder={selected}
          anchor={menuAnchor}
          onRename={rename}
          onDelete={requestDeletion}
          onClose={closeMenu}
        />
      )}
      {editor.editing && (
        <FolderNameField
          name={editor.name}
          busy={editor.busy}
          onName={editor.setName}
          onSave={() => void editor.submit()}
          onCancel={editor.cancel}
        />
      )}
      <p className="sr-only" role="status" aria-live="polite">
        {announcement && <span key={announcement.id}>{announcement.text}</span>}
      </p>
      {deletion.confirming && (
        <FolderDeletionDialog
          folder={deletion.confirming}
          busy={deletion.deleting}
          onConfirm={deletion.confirm}
          onClose={deletion.cancel}
        />
      )}
    </div>
  );
}

import { useEffect, useState } from 'react';

interface Clock {
  readonly elapsed: number;
}

/** Re-reads a running clock every second; the timer never animates. */
export function useElapsed(clock: Clock): number {
  const [, setTick] = useState(0);
  useEffect(() => {
    const timer = setInterval(() => setTick((tick) => tick + 1), 1_000);
    return () => clearInterval(timer);
  }, [clock]);
  return clock.elapsed;
}

export const LIST_ERROR = 'Impossible de charger vos réunions.';

interface WorkspaceErrors {
  recording: string;
  analysis: string;
  library: string;
  folders: string;
}

export interface WorkspaceNotice {
  message: string;
  retryList: boolean;
}

/** One notice at a time, the most urgent first; only the list can be retried from it. */
export function workspaceNotice(errors: WorkspaceErrors): WorkspaceNotice | null {
  if (errors.recording) return { message: errors.recording, retryList: false };
  if (errors.analysis) return { message: errors.analysis, retryList: false };
  if (errors.library) return { message: LIST_ERROR, retryList: true };
  if (errors.folders) return { message: errors.folders, retryList: false };
  return null;
}

import { useCallback, useState } from 'react';
import type { DesktopApi, Meeting } from '../shared/contracts';
import { resolveSelection } from './features/list/selection';
import { useFolders } from './features/list/use-folders';
import { useMeetings } from './features/list/use-meetings';
import { useAutoAnalysis } from './features/meetings/use-auto-analysis';
import type { CaptureOptions } from './features/recording/capture-session';
import { useRecording } from './features/recording/use-recording';
import { workspaceNotice } from './workspace-notice';

export type CardView = 'meeting' | 'settings';
type StartOptions = Omit<CaptureOptions, 'onFailure' | 'onEnded'>;

export function useWorkspace(api: DesktopApi) {
  const library = useMeetings(api);
  const folders = useFolders(api, library.update, library.refresh);
  const analysis = useAutoAnalysis(api, library.update);
  const [card, setCard] = useState<CardView>('meeting');
  const [chosenId, setChosenId] = useState<string | null>(null);
  const [newMeeting, setNewMeeting] = useState(false);
  const { run: runAutoAnalysis } = analysis;
  const updateMeeting = library.update;
  const onFinished = useCallback(
    (meeting: Meeting, autoAnalyze: boolean) => {
      updateMeeting(meeting);
      setChosenId(meeting.id);
      setCard('meeting');
      if (autoAnalyze) runAutoAnalysis(meeting);
    },
    [runAutoAnalysis, updateMeeting],
  );
  const recording = useRecording(api, onFinished);

  async function start(title: string, options: StartOptions, autoAnalyze: boolean) {
    analysis.clearError();
    await recording.start(title, options, autoAnalyze);
    setNewMeeting(false);
  }

  function select(id: string): void {
    setChosenId(id);
    setCard('meeting');
  }

  return {
    library,
    folders,
    recording,
    card,
    newMeeting,
    selected: resolveSelection(library.meetings, chosenId),
    notice: workspaceNotice({
      recording: recording.error,
      analysis: analysis.error,
      library: library.error,
      folders: folders.error,
    }),
    start,
    select,
    toggleSettings: () => setCard((current) => (current === 'settings' ? 'meeting' : 'settings')),
    showSettings: () => setCard('settings'),
    openNewMeeting: () => setNewMeeting(true),
    closeNewMeeting: () => setNewMeeting(false),
  };
}
